39vonMisesCalving::vonMisesCalving(std::shared_ptr<const Grid> grid,
40 std::shared_ptr<const rheology::FlowLaw> flow_law)
41 : StressCalving(grid, 2),
42 m_calving_threshold(m_grid, "vonmises_calving_threshold"),
43 m_flow_law(flow_law)
44{
45
46 if (m_config->get_flag("calving.vonmises_calving.use_custom_flow_law")) {
47 rheology::FlowLawFactory factory(m_config, grid->ctx()->enthalpy_converter());
48 m_flow_law = factory.create(m_config->get_string("calving.vonmises_calving.flow_law"),
49 m_config->get_number("calving.vonmises_calving.Glen_exponent"));
50 }
51
52 m_calving_rate.metadata().set_name("vonmises_calving_rate");
53 m_calving_rate.metadata(0)
54 .long_name("horizontal calving rate due to von Mises calving")
55 .units("m s^-1")
56 .output_units("m year^-1");
57
58 m_calving_threshold.metadata(0)
59 .long_name("threshold used by the 'von Mises' calving method")
60 .units("Pa")
61 .set_time_dependent(false); // no standard name
62}

64void vonMisesCalving::init() {
65
66 m_log->message(2,
67 "* Initializing the 'von Mises calving' mechanism...\n");
68
69 std::string threshold_file = m_config->get_string("calving.vonmises_calving.threshold_file");
70 double sigma_max = m_config->get_number("calving.vonmises_calving.sigma_max");
71
72 m_calving_threshold.set(sigma_max);
73
74 if (not threshold_file.empty()) {
75 m_log->message(2,
76 " Reading von Mises calving threshold from file '%s'...\n",
77 threshold_file.c_str());
78
79 m_calving_threshold.regrid(threshold_file, io::Default::Nil());
80 } else {
81 m_log->message(2,
82 " von Mises calving threshold: %3.3f Pa.\n", sigma_max);
83 }
84
85 // grid "squareness" criterion
86 const double eps = 1e-2;
87 if (fabs(m_grid->dx() - m_grid->dy()) / std::min(m_grid->dx(), m_grid->dy()) > eps) {
88 throw RuntimeError::formatted(PISM_ERROR_LOCATION,
89 "-calving vonmises_calving using a non-square grid cell is not implemented (yet);\n"
90 "dx = %f, dy = %f, relative difference = %f",
91 m_grid->dx(), m_grid->dy(),
92 fabs(m_grid->dx() - m_grid->dy()) / std::max(m_grid->dx(), m_grid->dy()));
93 }
94
95 m_strain_rates.set(0.0);
96}

98//! \brief Uses principal strain rates to apply the "von Mises" calving method
99/*!
100 See equation (4) in [@ref Morlighem2016].
101*/
102void vonMisesCalving::update(const array::CellType1 &cell_type,
103 const array::Scalar &ice_thickness,
104 const array::Vector1 &ice_velocity,
105 const array::Array3D &ice_enthalpy) {
106
107 using std::max;
108 using std::sqrt;
109 using std::pow;
110
111 // Distance (grid cells) from calving front where strain rate is evaluated
112 int offset = m_stencil_width;
113
114 // make a copy with a wider stencil
115 m_cell_type.copy_from(cell_type);
116
117 stressbalance::compute_2D_principal_strain_rates(ice_velocity, m_cell_type,
118 m_strain_rates);
119 m_strain_rates.update_ghosts();
120
121 array::AccessScope list{&ice_enthalpy, &ice_thickness, &m_cell_type, &ice_velocity,
122 &m_strain_rates, &m_calving_rate, &m_calving_threshold};
123
124 const double *z = ice_enthalpy.levels().data();
125
126 double glen_exponent = m_flow_law->exponent();
127
128 for (auto pt : m_grid->points()) {
129 const int i = pt.i(), j = pt.j();
130
131 // Find partially filled or empty grid boxes on the icefree ocean, which
132 // have floating ice neighbors after the mass continuity step
133 if (m_cell_type.ice_free_ocean(i, j) and m_cell_type.next_to_ice(i, j)) {
134
135 double
136 velocity_magnitude = 0.0,
137 hardness = 0.0;
138 // Average of strain-rate eigenvalues in adjacent floating grid cells.
139 double
140 eigen1 = 0.0,
141 eigen2 = 0.0;
142 {
143 int N = 0;
144 for (int p = -1; p < 2; p += 2) {
145 const int I = i + p * offset;
146 if (m_cell_type.icy(I, j)) {
147 velocity_magnitude += ice_velocity(I, j).magnitude();
148 {
149 double H = ice_thickness(I, j);
150 auto k = m_grid->kBelowHeight(H);
151 hardness += averaged_hardness(*m_flow_law, H, k, z, ice_enthalpy.get_column(I, j));
152 }
153 eigen1 += m_strain_rates(I, j).eigen1;
154 eigen2 += m_strain_rates(I, j).eigen2;
155 N += 1;
156 }
157 }
158
159 for (int q = -1; q < 2; q += 2) {
160 const int J = j + q * offset;
161 if (m_cell_type.icy(i, J)) {
162 velocity_magnitude += ice_velocity(i, J).magnitude();
163 {
164 double H = ice_thickness(i, J);
165 auto k = m_grid->kBelowHeight(H);
166 hardness += averaged_hardness(*m_flow_law, H, k, z, ice_enthalpy.get_column(i, J));
167 }
168 eigen1 += m_strain_rates(i, J).eigen1;
169 eigen2 += m_strain_rates(i, J).eigen2;
170 N += 1;
171 }
172 }
173
174 if (N > 0) {
175 eigen1 /= N;
176 eigen2 /= N;
177 hardness /= N;
178 velocity_magnitude /= N;
179 }
180 }
181
182 // [\ref Morlighem2016] equation 6
183 const double effective_tensile_strain_rate = sqrt(0.5 * (pow(max(0.0, eigen1), 2) +
184 pow(max(0.0, eigen2), 2)));
185 // [\ref Morlighem2016] equation 7
186 const double sigma_tilde = sqrt(3.0) * hardness * pow(effective_tensile_strain_rate,
187 1.0 / glen_exponent);
188
189 // Calving law [\ref Morlighem2016] equation 4
190 m_calving_rate(i, j) = velocity_magnitude * sigma_tilde / m_calving_threshold(i, j);
191
192 } else { // end of "if (ice_free_ocean and next_to_floating)"
193 m_calving_rate(i, j) = 0.0;
194 }
195 } // end of loop over grid points
196}
