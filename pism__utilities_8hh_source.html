93/*! Helper template function for computing set unions.
94 * Ensures that elements of a take precedence. For example, if
95 *
96 * a = {{1, 2}, {3, 4}}
97 * b = {{1, 4}, {5, 6}}
98 *
99 * combine(a, b) will use the pair {1, 2} from a, not {1, 4} from b.
100 *
101 * This behavior relies on the fact that std::map::insert({a, b}) is a no-op if a key equivalent to
102 * a is already present.
103 *
104 * This is similar to a set union, but it is not symmetric. (I would expect set_union(a, b) to be
105 * the same as set_union(b, a)).
106 */
107template<typename T>
108T combine(const T &a, const T&b) {
109 T result = a;
110 for (const auto &element : b) {
111 result.insert(element);
112 }
113 return result;
114}
