28//! Initialize checkpointing (snapshot-on-wallclock-time) mechanism.
29void IceModel::init_checkpoints() {
30
31 m_checkpoint_filename = m_config->get_string("output.checkpoint.file");
32
33 if (m_checkpoint_filename.empty()) {
34 m_checkpoint_filename = filename_add_suffix(m_output_filename, "_checkpoint", "");
35 }
36
37 m_checkpoint_vars = output_variables(m_config->get_string("output.checkpoint.size"));
38 m_last_checkpoint_time = 0.0;
39
40 {
41 m_checkpoint_file_contents = pism::combine(common_metadata(), state_variables());
42 m_checkpoint_file_contents =
43 pism::combine(m_checkpoint_file_contents, state_variables_diagnostics(m_checkpoint_vars));
44 m_checkpoint_file_contents =
45 pism::combine(m_checkpoint_file_contents, state_variables_diagnostics(m_spatial_vars));
46 m_checkpoint_file_contents =
47 pism::combine(m_checkpoint_file_contents, diagnostic_variables(m_checkpoint_vars));
48 }
49}

51//! Write a checkpoint (i.e. an intermediate result of a run).
52/*!
53 * Returns `true` if PISM has to stop, `false` otherwise.
54 */
55bool IceModel::write_checkpoint() {
56
57 double checkpoint_interval = m_config->get_number("output.checkpoint.interval");
58
59 double wall_clock_hours = pism::wall_clock_hours(m_grid->com, m_start_time);
60
61 if (wall_clock_hours - m_last_checkpoint_time < checkpoint_interval) {
62 return false;
63 }
64
65 const Profiling &profiling = m_ctx->profiling();
66
67 m_last_checkpoint_time = wall_clock_hours;
68
69 // create a history string:
70
71 m_log->message(2,
72 " [%s] Saving a checkpoint to '%s' (%1.3f hours after the beginning of the run)\n",
73 timestamp(m_grid->com).c_str(), m_checkpoint_filename.c_str(), wall_clock_hours);
74
75 double checkpoint_start_time = get_time(m_grid->com);
76 profiling.begin("io.checkpoint");
77 {
78 OutputFile file(m_output_writer, m_checkpoint_filename);
79 // Ensure that the checkpoint file is closed to force PISM to open a new file every
80 // time we write a checkpoint, moving the old file aside if it exists.
81 file.close();
82
83 {
84 // define time dimension *without* time bounds
85 define_time(file);
86 define_variables(file, m_checkpoint_file_contents);
87 }
88
89 {
90 io::write_config(*m_config, "pism_config", file);
91 file.append_time(m_time->current());
92 write_state(file);
93 write_state_diagnostics(file, m_checkpoint_vars);
94 write_diagnostics(file, m_checkpoint_vars);
95 write_run_stats(file);
96 }
97 }
98 profiling.end("io.checkpoint");
99 double checkpoint_end_time = get_time(m_grid->com);
100
101 // Also flush time-series:
102 scalar_diagnostics_flush_buffers();
103
104 m_log->message(2,
105 " [%s] Done saving a checkpoint in %f seconds (%f minutes).\n",
106 timestamp(m_grid->com).c_str(),
107 checkpoint_end_time - checkpoint_start_time,
108 (checkpoint_end_time - checkpoint_start_time) / 60.0);
109
110 return m_config->get_flag("output.checkpoint.exit");
111}
