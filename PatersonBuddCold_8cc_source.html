51// Rather than make this part of the base class, we just check at some reference values.
52bool FlowLawIsPatersonBuddCold(const FlowLaw &flow_law, const Config &config,
53 std::shared_ptr<EnthalpyConverter> EC) {
54 static const struct {
55 double s, E, p, gs;
56 } v[] = { { 1e3, 223, 1e6, 1e-3 },
57 { 450000, 475000, 500000, 525000 },
58 { 5e4, 268, 5e6, 3e-3 },
59 { 1e5, 273, 8e6, 5e-3 } };
60
61 double n = config.get_number("stress_balance.sia.Glen_exponent");
62 PatersonBuddCold cpb(n, config, EC); // This is unmodified cold Paterson-Budd
63
64 for (int i=0; i<4; i++) {
65 const double left = flow_law.flow(v[i].s, v[i].E, v[i].p, v[i].gs),
66 right = cpb.flow(v[i].s, v[i].E, v[i].p, v[i].gs);
67 if (fabs((left - right)/left)>1.0e-15) {
68 return false;
69 }
70 }
71 return true;
72}
