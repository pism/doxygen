46static void set_config_defaults(Config &config) {
47 config.set_number("grid.Lx", 750); // in km
48 config.set_number("grid.Ly", 750); // in km
49 config.set_string("grid.periodicity", "none");
50 config.set_string("grid.registration", "corner");
51 config.set_string("stress_balance.sia.flow_law", "pb");
52
53 config.set_string("energy.model", "cold");
54
55 // Set sea level elevation to -1e4 meters to remove ocean interaction
56 config.set_number("sea_level.constant.value", -1e4);
57
58 // purely SIA, and E=1
59 config.set_number("stress_balance.sia.enhancement_factor", 1.0);
60
61 // none use bed smoothing & bed roughness parameterization
62 config.set_number("stress_balance.sia.bed_smoother.range", 0.0);
63
64 // basal melt does not change computation of mass continuity or vertical velocity:
65 config.set_flag("geometry.update.use_basal_melt_rate", false);
66
67 // Make bedrock thermal material properties into ice properties. Note that
68 // zero thickness bedrock layer is the default, but we want the ice/rock
69 // interface segment to have geothermal flux applied directly to ice without
70 // jump in material properties at base.
71 config.set_number("energy.bedrock_thermal.density",
72 config.get_number("constants.ice.density"));
73 config.set_number("energy.bedrock_thermal.conductivity",
74 config.get_number("constants.ice.thermal_conductivity"));
75 config.set_number("energy.bedrock_thermal.specific_heat_capacity",
76 config.get_number("constants.ice.specific_heat_capacity"));
77
78 // no sliding + SIA
79 config.set_string("stress_balance.model", "sia");
80}
81} // namespace eismint2

85void set_config_defaults(Config &config, char testname) {
86 // This sets the defaults for each test; command-line options can override this.
87
88 config.set_number("grid.Mx", 61);
89 config.set_number("grid.My", 61);
90 config.set_number("grid.Lx", 1); // in km
91 config.set_number("grid.Ly", 1); // in km
92
93 // use the cell corner grid registration
94 config.set_string("grid.registration", "corner");
95
96 // use the non-periodic grid:
97 config.set_string("grid.periodicity", "none");
98
99 // quadratic spacing is the default
100 config.set_string("grid.ice_vertical_spacing", "quadratic");
101
102 switch (testname) {
103 case 'A':
104 case 'B':
105 case 'H':
106 // use 2400km by 2400km by 4000m rectangular domain
107 config.set_number("grid.Lx", 1200); // in km
108 config.set_number("grid.Ly", 1200); // in km
109 config.set_number("grid.Lz", 4000); // in m
110 break;
111 case 'C':
112 case 'D':
113 // use 2000km by 2000km by 4000m rectangular domain
114 config.set_number("grid.Lx", 1000); // in km
115 config.set_number("grid.Ly", 1000); // in km
116 config.set_number("grid.Lz", 4000); // in m
117 break;
118 case 'F':
119 case 'G':
120 case 'L':
121 // use 1800km by 1800km by 4000m rectangular domain
122 config.set_number("grid.Lx", 900); // in km
123 config.set_number("grid.Ly", 900); // in km
124 config.set_number("grid.Lz", 4000); // in m
125 break;
126 case 'K':
127 case 'O':
128 // use 2000km by 2000km by 4000m rectangular domain, but make truely periodic
129 config.set_number("grid.Mbz", 2);
130 config.set_number("grid.Lbz", 1000); // in m
131 config.set_number("grid.Lx", 1000); // in km
132 config.set_number("grid.Ly", 1000); // in km
133 config.set_number("grid.Lz", 4000); // in m
134 config.set_string("grid.periodicity", "xy");
135 break;
136 case 'V':
137 config.set_number("grid.My", 3);// it's a flow-line setup
138 config.set_number("grid.Lx", 500); // 500 km long
139 config.set_string("grid.periodicity", "y");
140 break;
141 default:
142 throw RuntimeError(PISM_ERROR_LOCATION, "desired test not implemented\n");
143 }
144}

146//! Allocate the verification mode context. Uses ColdEnthalpyConverter.
147std::shared_ptr<Context> context(MPI_Comm com, const std::string &prefix,
148 char testname) {
149 // unit system
150 auto sys = std::make_shared<units::System>();
151
152 // logger
153 auto logger = std::make_shared<Logger>(com, 1);
154
155 // configuration parameters
156 auto config = config_from_options(com, sys);
157
158 set_config_defaults(*config, testname);
159 set_config_from_options(*config);
160 config->resolve_filenames();
161
162 logger->set_threshold(static_cast<int>(config->get_number("output.runtime.verbosity")));
163
164 print_config(*logger, 3, *config);
165
166 auto time = std::make_shared<Time>(com, config, *logger, sys);
167
168 auto EC = std::make_shared<ColdEnthalpyConverter>(*config);
169
170 return std::make_shared<Context>(com, sys, config, EC, time, logger, prefix);
171}

173std::shared_ptr<Grid> grid(std::shared_ptr<Context> ctx) {
174 auto config = ctx->config();
175
176 auto input_file_name = config->get_string("input.file");
177
178 if (config->get_flag("input.bootstrap")) {
179 throw RuntimeError(PISM_ERROR_LOCATION, "PISM does not support bootstrapping in verification mode");
180 }
181
182 if (not input_file_name.empty()) {
183 auto r = grid::string_to_registration(config->get_string("grid.registration"));
184
185
186 File input_file(ctx->com(), input_file_name, pism::io::PISM_NETCDF3, pism::io::PISM_READONLY);
187 // get grid from a PISM input file
188 return Grid::FromFile(ctx, input_file, { "enthalpy", "temp" }, r);
189 }
190
191 return Grid::FromOptions(ctx);
192}

195int main(int argc, char *argv[]) {
196
197 MPI_Comm com = MPI_COMM_WORLD;
198 petsc::Initializer petsc(argc, argv, help);
199
200 com = PETSC_COMM_WORLD;
201
202 int exit_code = 0;
203 try {
204 // Note: EISMINT II experiments G and H are not supported.
205 auto eisII = options::Keyword("-eisII",
206 "EISMINT II experiment name",
207 "A,B,C,D,E,F,I,J,K,L", "A");
208
209 auto verification_test =
210 options::Keyword("-test", "Specifies PISM verification test", "A,B,C,D,F,G,H,K,L,V", "A");
211
212 if (eisII.is_set() and verification_test.is_set()) {
213 throw RuntimeError::formatted(PISM_ERROR_LOCATION, "please set -test or -eisII (not both)");
214 }
215
216 std::shared_ptr<Context> ctx;
217 if (verification_test.is_set()) {
218 char test = verification_test.value()[0];
219 ctx = verification::context(com, "pism", test);
220 } else {
221 ctx = context_from_options(com, "pism");
222 }
223
224 auto log = ctx->log();
225 auto config = ctx->config();
226
227 if (eisII.is_set()) {
228 // set defaults:
229 eismint2::set_config_defaults(*config);
230
231 // process -config_override
232 auto overrides = std::make_shared<NetCDFConfig>("pism_overrides", ctx->unit_system());
233
234 options::String override_filename("-config_override", "Config override file name");
235
236 if (override_filename.is_set()) {
237 overrides->read(com, override_filename);
238 config->import_from(*overrides);
239 }
240
241 // process command-line options
242 set_config_from_options(*config);
243 config->resolve_filenames();
244 }
245
246 if (options::Bool("-version", "print PISM version and stop")) {
247 log->message(1, pism::version());
248 return 0;
249 }
250
251 if (not(verification_test.is_set() or eisII.is_set()) and
252 config->get_string("input.file").empty()) {
253 throw RuntimeError(PISM_ERROR_LOCATION,
254 "Configuration parameter 'input.file' cannot be empty");
255 }
256
257 print_config(*ctx->log(), 3, *config);
258
259 std::string usage =
260 " pism -i IN.nc [-bootstrap] [-regional] [OTHER PISM & PETSc OPTIONS]\n"
261 "where:\n"
262 " -i IN.nc is input file in NetCDF format: contains PISM-written model state\n"
263 " -bootstrap enable heuristics to produce an initial state from an incomplete input\n"
264 " -regional enable \"regional mode\"\n"
265 " -eisII [experiment] enable EISMINT II mode\n"
266 " -test [verification_test] enable verification mode\n"
267 "notes:\n"
268 " * option -i is required\n"
269 " * if -bootstrap is used then also '-Mx A -My B -Mz C -Lz D' are required\n";
270 {
271 bool done = maybe_show_usage(*log, "PISM (basic evolution run mode)", usage);
272 if (done) {
273 return 0;
274 }
275 }
276
277 options::String profiling_log = options::String("-profile",
278 "Save detailed profiling data to a file.");
279
280 if (profiling_log.is_set()) {
281 ctx->profiling().start();
282 }
283
284 std::shared_ptr<Grid> grid;
285 std::shared_ptr<IceModel> model;
286 std::shared_ptr<IceCompModel> verification_model;
287
288 if (verification_test.is_set()) {
289 char test = verification_test.value()[0];
290 grid = verification::grid(ctx);
291
292 verification_model = std::make_shared<IceCompModel>(grid, ctx, test);
293 model = verification_model;
294 } else {
295 grid = Grid::FromOptions(ctx);
296
297 if (options::Bool("-regional", "enable regional (outlet glacier) mode")) {
298 model = std::make_shared<IceRegionalModel>(grid, ctx);
299 } else if (eisII.is_set()) {
300 char experiment = eisII.value()[0];
301
302 model = std::make_shared<IceEISModel>(grid, ctx, experiment);
303 } else {
304 model = std::make_shared<IceModel>(grid, ctx);
305 }
306 }
307
308 auto list_type = options::Keyword("-list_diagnostics",
309 "List available diagnostic quantities and stop.",
310 "all,spatial,scalar,json",
311 "all");
312
313 if (list_type.is_set()) {
314 auto report = DIAG_NONE;
315 auto value = list_type.value();
316 if (value == "spatial") {
317 report = DIAG_SPATIAL;
318 } else if (value == "scalar") {
319 report = DIAG_SCALAR;
320 } else if (value == "json") {
321 report = DIAG_JSON;
322 } else {
323 report = DIAG_ALL;
324 }
325 model->init(report);
326
327 } else {
328 model->init(DIAG_NONE);
329
330 auto termination_reason = model->run();
331
332 switch (termination_reason) {
333 case PISM_CHEKPOINT:
334 {
335 exit_code = static_cast<int>(config->get_number("output.checkpoint.exit_code"));
336 log->message(2, "... stopping (exit_code=%d) after saving the checkpoint file\n",
337 exit_code);
338 break;
339 }
340 case PISM_SIGNAL:
341 case PISM_DONE:
342 {
343 log->message(2, "... done with the run\n");
344 model->write_final_output();
345 exit_code = 0;
346
347 if (verification_model and
348 not options::Bool("-no_report", "do not print the error report")) {
349 verification_model->reportErrors();
350 }
351 break;
352 }
353 }
354 }
355 print_unused_parameters(*log, 3, *config);
356
357 if (profiling_log.is_set()) {
358 ctx->profiling().report(profiling_log);
359 }
360 }
361 catch (...) {
362 handle_fatal_errors(com);
363 exit_code = 1;
364 }
365
366 return exit_code;
367}
