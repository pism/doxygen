32/**
33 * Returns 0 if `name` is a name of a supported calendar, 1 otherwise.

35inline bool pism_is_valid_calendar_name(const std::string &name) {
36 // Calendar names from the CF Conventions document (except the
37 // 366_day (all_leap)):
38 return set_member(name, {"standard", "gregorian", "proleptic_gregorian",
39 "noleap", "365_day", "julian", "360_day"});
40}

42//! \brief Time management class.

44 * This is to make it possible to switch between different implementations.
