97//! \brief Distribute the input velocity throughout the column.
98/*!
99 * Things to update:
100 * - 3D-distributed horizontal velocity
101 * - maximum horizontal velocity
102 * - diffusive ice flux
103 * - maximum diffusivity
104 * - strain heating (strain_heating)
105 */
106void ConstantInColumn::update(const array::Vector &sliding_velocity,
107 const Inputs &inputs,
108 bool full_update) {
109
110 (void) inputs;
111
112 if (not full_update) {
113 return;
114 }
115
116 // horizontal velocity and its maximum:
117 array::AccessScope list{&m_u, &m_v, &sliding_velocity};
118
119 for (auto p : m_grid->points()) {
120 const int i = p.i(), j = p.j();
121
122 m_u.set_column(i,j, sliding_velocity(i,j).u);
123 m_v.set_column(i,j, sliding_velocity(i,j).v);
124 }
125
126 // Communicate to get ghosts (needed to compute w):
127 m_u.update_ghosts();
128 m_v.update_ghosts();
129
130 // diffusive flux and maximum diffusivity
131 m_diffusive_flux.set(0.0);
132 m_D_max = 0.0;
133}
