26/*!
27 * Update a `model` by asking it to perform time-stepping from the current time to one year in the
28 * future (or as far as the time step restriction allows).
29 *
30 * This is sometimes necessary during initialization, but should be avoided if possible.
31 */
32template<class M, class In>
33void init_step(M *model, const In &inputs, const Time& time) {
34 const double
35 now = time.current(),
36 one_year_from_now = time.increment_date(now, 1.0);
37
38 // Take a one year long step if we can.
39 MaxTimestep max_dt(one_year_from_now - now);
40
41 max_dt = std::min(max_dt, model->max_timestep(now));
42
43 // Do not take time-steps shorter than 1 second
44 if (max_dt.value() < 1.0) {
45 max_dt = MaxTimestep(1.0);
46 }
47
48 assert(max_dt.finite());
49
50 model->update(inputs, now, max_dt.value());
51}
