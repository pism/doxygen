32//! \brief Computes result = x + alpha * y, where x, y, and z are 2D
33//! Arrays (scalar or vector).

36template <class V>
37void add(const V &x, double alpha, const V &y, V &result, bool scatter = true) {
38
39 array::AccessScope list{ &x, &y, &result };
40 for (auto p : result.grid()->points()) {
41 const int i = p.i(), j = p.j();
42
43 result(i, j) = x(i, j) + y(i, j) * alpha;
44 }
45
46 if (scatter) {
47 result.update_ghosts();
48 }
49
50 result.inc_state_counter();
51}

63
64 if (scatter) {

66 }
67
