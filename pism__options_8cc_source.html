49//! @brief In a single call a driver program can provide a usage string to
50//! the user and check if required options are given, and if not, end.
51bool maybe_show_usage(const Logger &log, const std::string &execname,
52 const std::string &usage) {
53 const bool keep_running = false;
54 const bool terminate = true;
55
56 log.message(2, "%s %s\n", execname.c_str(), pism::revision);
57
58 if (options::Bool("-version", "print PISM version and stop")) {
59 log.message(2, pism::version());
60 return terminate;
61 }
62
63 if (options::Bool("-usage", "print PISM usage")) {
64 show_usage(log, execname, usage);
65 return terminate;
66 }
67
68 // show usage message with -help, but don't stop
69 if (options::Bool("-help", "print help on all options")) {
70 show_usage(log, execname, usage);
71 }
72 return keep_running;
73}
