66void Runoff_SMB::mass_flux(double delta_T, array::Scalar &result) const {
67
68 // short-cuts, just to make the formula below easier to read
69 double
70 a = m_temp_to_runoff_a,
71 B = m_runoff_to_ocean_melt_b,
72 alpha = m_runoff_to_ocean_melt_power_alpha,
73 beta = m_runoff_to_ocean_melt_power_beta,
74 scale_factor = 1.0;
75
76 /*
77 This parameterization only works if delta_T > 0 because
78 negative numbers cannot be raised to a fractional power
79 so we do not scale the forcing if delta_T < 0
80 */
81 if (delta_T > 0.0) {
82 scale_factor = 1 + B * pow(a * delta_T, alpha) * pow(delta_T, beta);
83 }
84 result.scale(scale_factor);
85}
