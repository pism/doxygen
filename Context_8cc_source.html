121std::shared_ptr<Context> context_from_options(MPI_Comm com,
122 const std::string &prefix) {
123 // unit system
124 auto sys = std::make_shared<units::System>();
125
126 // configuration parameters
127 auto config = config_from_options(com, sys);
128
129 return context_from_config(com, config, prefix);
130}

132std::shared_ptr<Context> context_from_config(MPI_Comm com,
133 std::shared_ptr<Config> config,
134 const std::string &prefix) {
135 // unit system
136 auto sys = config->unit_system();
137
138 // logger
139 auto logger = std::make_shared<Logger>(com, 1);
140
141 logger->set_threshold(static_cast<int>(config->get_number("output.runtime.verbosity")));
142
143 // time manager
144 auto time = std::make_shared<Time>(com, config, *logger, sys);
145
146 // enthalpy converter
147 auto EC = std::make_shared<EnthalpyConverter>(*config);
148
149 return std::make_shared<Context>(com, sys, config, EC, time, logger, prefix);
150}
